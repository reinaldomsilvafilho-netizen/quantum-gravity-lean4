\section*{Scope and Organization}
This dictionary collects the symbols, terms and principal statements of the 13 chapters of the book. The chapters are grouped in five parts: I.~Foundations of Functional Algebra, Hypertensors, and Geometric Flows (Ch.~1--2); II.~Continuous Simplicial Geometry and Fractional Calculus on Pascal Simplices (Ch.~3--6); III.~Non-Euclidean Minimax Extrinsic Curvature and Relativistic Foliations (Ch.~7--10); IV.~Emergent Spacetime: Established Results, Conjectures, and Heuristics (Ch.~11--12); V.~Observational Signatures: Order-of-Magnitude Estimates (Ch.~13).

The dictionary has four parts: (1) a symbol lexicon; (2) core formulations, each marked as \emph{proved}, \emph{classical (cited)}, \emph{conjectural}, \emph{numerical observation} or \emph{phenomenological ansatz}, in accordance with the chapters; (3) the verification status; and (4) a cross-disciplinary table, whose entries are analogies rather than equivalences. Results are referred to by chapter and name; their numbers, proofs and verification status are listed in the Results Index at the end of the book. Where this dictionary and a chapter differ, the chapter is authoritative. Some letters carry different meanings in different chapters (for instance $\mathcal{A}$, $\mathcal{L}$, $\chi$ and $\rho$); the chapter column then fixes the meaning.

\section{Core Mathematical Formulations and Structural Theorems}

\begin{conceptbox}{{1. Simplicial Fractional Laplacian ($-\Delta_{\Delta_m}^\alpha$) \& $A_{m-1}$ Root Metric (Ch. 3, 4, 12)}}
\textbf{Definition:} bounded self-adjoint operator on $L^2(\mathbb{R}^{m-1})$, the symmetric second difference averaged against the continuous multinomial kernel on $\Delta_{m-1}(\alpha)$:
$$\Delta_{\Delta_m}^\alpha u(\mathbf{x}) = \frac{1}{2\alpha^2 I_m(\alpha)}\int_{\Delta_{m-1}(\alpha)} \binom{\alpha}{\mathbf{y}} \big[u(\mathbf{x}+\mathbf{y}) - 2u(\mathbf{x}) + u(\mathbf{x}-\mathbf{y})\big]\, d\mathbf{y}.$$
\textbf{Long-Wavelength Dispersion (proved, Ch.~3, Fourier symbol theorem; restated in Ch.~4):} the symbol lies in $[0,2/\alpha^2]$, and
$$\sigma_{\Delta_m}^\alpha(\mathbf{k}) = \frac{1}{2\alpha^2} \mathbf{k}^T \mathbf{M}_2(\alpha) \mathbf{k} + \mathcal{O}(|\mathbf{k}|^4), \qquad \mathbf{M}_2 = a\,\mathrm{Id} + b\,\mathbf{1}\mathbf{1}^T.$$
The form $\mathbf{k}^T\mathbf{M}_2\mathbf{k}$ is invariant under permutations of $k_1, \dots, k_{m-1}$ and under $\mathbf{k} \mapsto -\mathbf{k}$, but not under the full group $S_m$. The $A_{m-1}$ Gram form $\frac{1}{2m\alpha}\mathbf{k}^T\mathbf{G}_{m-1}\mathbf{k}$ is not the leading term; it would arise only under the constraint $\sum_j k_j = 0$. The Cartan matrix does appear, exactly, in the Hessian of $\log\binom{x}{\mathbf{t}}$ at the barycenter written in simple-root coordinates (Ch.~12, barycentric Hessian proposition), as a consequence of the $S_m$ symmetry of the multinomial.\\
\textbf{Lattice only:} the closed form $(\frac1m\sum e^{-ik_j})^\alpha$ and the semigroup law hold for the lattice (Gr\"unwald--Letnikov) construction, not the continuous kernel; the operator is bounded, so there is no Weyl law at $\lambda \to \infty$.
\end{conceptbox}

\begin{conceptbox}{{2. Inter-Dimensional Trace Operator $\mathcal{R}_{m \to n}^\alpha$ (Ch. 5)}}
\textbf{Definition:} $\mathcal{R}_{m \to n}^\alpha f(\mathbf{z}) = \int_{\Delta_m(\alpha)} \mathcal{K}_\alpha(\mathbf{y})\, f(\mathbf{P}^+\mathbf{z} + \mathbf{y})\, d\mathbf{y}$, i.e.\ convolution with the reflected kernel followed by restriction to the $n$-plane $\operatorname{ran}\mathbf{P}^+$. In the Fourier domain,
$$\widehat{\mathcal{R}_{m \to n}^\alpha f}(\boldsymbol{\xi}) = (2\pi)^{n-m}\sqrt{\det(\mathbf{P}\mathbf{P}^T)} \int_{\operatorname{ker}(\mathbf{P})} \widehat{f}(\mathbf{P}^T \boldsymbol{\xi} + \boldsymbol{\eta})\, \widehat{\mathcal{K}}_\alpha(-\mathbf{P}^T \boldsymbol{\xi} - \boldsymbol{\eta})\, d\boldsymbol{\eta}.$$
\textbf{Trace estimate (proved, Ch.~5):} $\mathcal{R}_{m\to n}^\alpha : H^s(\mathbb{R}^m) \to H^{s+\gamma-\frac{m-n}{2}}(\mathbb{R}^n)$ for $s + \gamma > \frac{m-n}{2}$, where $\gamma$ is the Fourier decay rate of the kernel. For the Beta-kernel $\gamma = 1$ exactly, independently of $\alpha$.\\
\textbf{Inversion (proved, Ch.~5):} a single trace is never injective, but the family of traces $g_\theta$ over all $n$-planes $\theta$ determines the smoothed field $h(\mathbf{x}) = g_\theta(\mathbf{P}_\theta\mathbf{x})$ ($\mathbf{x} \in \theta$), and $f = \mathcal{F}^{-1}[\widehat{h}(\mathbf{k})/\widehat{\mathcal{K}}_\alpha(-\mathbf{k})]$ for $f \in \mathcal{S}$, the quotient being defined almost everywhere because the zero set of $\widehat{\mathcal{K}}_\alpha$ is Lebesgue-null ($\widehat{\mathcal{K}}_\alpha$ is entire with $\widehat{\mathcal{K}}_\alpha(0) = 1$). No hypothesis on the zeros of $\widehat{\mathcal{K}}_\alpha$ is needed. The data are restrictions, not $n$-plane integrals, so filtered backprojection does not apply; the deconvolution is ill-posed of order one.\\
\textbf{Barycentric contraction (proved, Ch.~5):} $\operatorname{dist}_\Delta(\Phi_\alpha\mathbf{c}_1, \Phi_\alpha\mathbf{c}_2) \le \frac{\alpha}{\alpha+m}\operatorname{dist}_\Delta(\mathbf{c}_1,\mathbf{c}_2)$, with a sharp constant.\\
\textbf{Note:} in general $\gamma \ne \alpha$.
\end{conceptbox}

\begin{conceptbox}{{3. Minimax Curvature $\kappa^*$, Curvature Loci \& $C^{1,1}$ Regularity (Ch. 7, 8)}}
\textbf{Definition:} $\kappa^*(\mathcal{C}) = \inf_{M \in \mathcal{C}} \operatorname{ess\,sup}_{x \in M} \|\II_M(x)\|_{\mathrm{op}, g}$. The flat, saturated, transition and contact loci are defined by the value of $\|\II\|_{\mathrm{op}}$; positivity of the saturated locus is a conjecture (Ch.~7).
\textbf{Proved (Ch.~7):} existence of $C^{1,1}$ minimizers for curves; contact principle $\|\II_M\| \ge \lambda_k(h_{\partial\Omega})$ (only obstacle boundaries curving towards $\Omega$ constrain); planar U-turn bound $2/w$; knot gap $\kappa^*_K(L) \ge 4\pi/L = \kappa^*_{\mathrm{Imm}}(L) + 2\pi/L$; regularity invariance $\kappa^*_{C^r} = \kappa^*_{C^{1,1}}$ for curves without obstacles; at the threshold length $V_0$ the $C^r$ class can be empty; dimensional monotonicity; Gauss--Bonnet floor $\kappa \ge \sqrt{2\pi\chi/A}$ for closed surfaces with $\chi > 0$. \textbf{Conjectural:} regularity invariance with obstacles for $V > V_0$; $C^{1,1}$ as the optimal class in general; discrete $\Gamma$-convergence.
\textbf{Global search (Ch.~9):} with a length bound only finitely many homotopy classes are admissible (proved); $\Gamma$-convergence of the spline discretization is a conjecture.
\textbf{ADM consequence (proved, Ch.~8 pointwise-bounds corollary; Ch.~12 pointwise constraint bounds):} on a slice with $\|K\|_{\mathrm{op}} \le \kappa$, $K_{ij}K^{ij} \le 3\kappa^2$ and $16\pi G\rho - 6\kappa^2 \le R^{(\gamma)} - 2\Lambda \le 16\pi G\rho + 2\kappa^2$. Whether slices with finite $\kappa^*$ exist and avoid singularities is \emph{not} established.
\end{conceptbox}

\begin{conceptbox}{{4. Winding Classes, Slingshot, and Jordan Loops (Ch. 8, 9, 12)}}
\textbf{Topology (proved, Ch.~8):} $\mathbb{R}^3$ minus disjoint balls is simply connected, so winding numbers about horizons are homotopy invariants only for motion confined to a plane, where $\pi_1 \cong \mathbb{F}_m$.
\textbf{Planar slingshot (conjecture, Ch.~8):} for planar worldlines, some winding classes $W = \pm 1$ reduce the minimax acceleration relative to $W = 0$.
\textbf{Chronology (proved, Ch.~12):} every timelike curve in a chronological spacetime (one without closed timelike curves, for instance a stably causal one) is injective. \textbf{Jordan-loop sector of LQG (conjecture, Ch.~12):} a regularization preserving Jordan-loop states with a closing constraint algebra exists. Chronology does not restrict the spacelike loops of LQG.
\end{conceptbox}

\begin{conceptbox}{{5. Graphon Ricci Flow and Condensation (Ch. 2, 12)}}
\textbf{Flow:} $\partial_t W = 2\kappa_W W$ on graphon kernels ($W$ a connection strength). Ollivier curvature is bounded below ($\kappa \ge -2$ in the graph setting).
\textbf{Proved (Ch.~12, exponential decay proposition):} if $\kappa_{W(s)} \le -c < 0$ on a set $B$ of positive measure for all $s \in [0,T]$, then $0 \le W(t) \le W_0 e^{-2ct}$ on $B$; hence for $\delta \in (0, \|W_0\|_{L^\infty(B)})$, $W(t) \le \delta$ for \emph{almost every} point of $B$ and all $t \in [t_\delta, T]$, $t_\delta = \frac{1}{2c}\ln(\|W_0\|_{L^\infty(B)}/\delta)$. The decay is exponential, not a finite-time pinch-off (cf.\ Ch.~2). With the discrete metric $\kappa_W = 1 - \tfrac12\|m_x - m_y\|_{L^1} \in [0,1]$, so the hypothesis never holds there.
\textbf{Conjecture:} condensation of the remaining clusters to a smooth 4-geometry, from a kernel with positive degree. A graphon carries no differentiable structure or metric tensor, so the target needs a correspondence between kernels and manifolds; trees, branched polymers and bounded-degree triangulations converge in cut norm to the zero kernel, where the flow is not defined.
\end{conceptbox}

\begin{conceptbox}{{6. Running Spectral Dimension $d_s = 2 \to 4$ (Ch. 12, 13)}}
\textbf{Proved:} for the symbol $|k|^{2\alpha}$ on $\mathbb{R}^m$, $d_s = m/\alpha$ (constant). For the two-scale symbol $k^2 + \ell^2 k^4$ in four dimensions,
$$d_s(\tau) = 1 - \frac{\tau}{2\ell^2} + \left[ 1 - \frac{\sqrt{\pi \tau}}{2\ell} e^{\tau/4\ell^2} \operatorname{erfc}\!\left( \frac{\sqrt{\tau}}{2\ell} \right) \right]^{-1}, \qquad d_s: 2 \ (\tau \to 0) \to 4 \ (\tau \to \infty).$$
\textbf{Context:} a closed form for a behaviour known from CDT fits (Sotiriou--Visser--Weinfurtner 2011) and Ho\v{r}ava--Lifshitz gravity (Ho\v{r}ava 2009). Its relation to UV finiteness is heuristic. \textbf{Scope (Ch.~12):} the symbol is assumed, with $\ell$ free; no translation-invariant operator with non-negative kernel and potential, nor a continuous refinement limit of such operators, has it. In Lorentzian signature it carries a ghost (residue $-1$); with a positive K\"all\'en--Lehmann weight, under the hypotheses of Ch.~12, $d_s \ge 4$ at every scale (Weinberg; Belenchia et al.).
\end{conceptbox}

\begin{conceptbox}{{7. Graviton Dispersion \& Tensor-Tilt Running (Ch. 12, 13) --- phenomenological ansatz}}
\textbf{Relations:} $\omega^2 = c^2 k^2 (1 + \xi \ell_\ast^2 k^2)$; $d_s(p) = 2 + \frac{2}{1 + (p/M_\ast)^2}$ (two-point Pad\'e interpolation with the limits of Box~6) and $\alpha_t = \frac{1}{2}(d_s(p) - 4) = -\frac{(p/M_\ast)^2}{1 + (p/M_\ast)^2}$, where $p = H_{\mathrm{inf}}$ is the physical momentum of the tensor mode at horizon exit.\\
\textbf{Magnitudes (Ch.~12, 13):} for $\ell_\ast = \ell_P$ and $\xi = 1/2$, $|\Delta t_{\mathrm{disp}}| = \frac{6\pi^2 \xi \ell_\ast^2}{c^3} D_2(z) (f_2^2 - f_1^2) \approx 6 \times 10^{-62}$--$1.2 \times 10^{-60}$\,s for $10$\,Hz--$1$\,kHz and redshifts $1$--$8$, $56$--$57$ orders of magnitude below a $10^{-4}$\,s timing benchmark for ET/CE. With $M_\ast = M_P$ and $H_{\mathrm{inf}} \le 4.71 \times 10^{13}$\,GeV, $|\alpha_t| \approx (H_{\mathrm{inf}}/M_P)^2 \le 1.5 \times 10^{-11}$ ($3.74 \times 10^{-10}$ for $M_\ast = M_{\mathrm{red}}$). No observable signature follows at the Planck scale. In a $z = 3$ realization weak coupling requires $M_\ast \lesssim 10^{15}$--$10^{16}$\,GeV, so $M_\ast \ne M_P$; if the quartic term is universal with unit coefficient, photon bounds (Crab Nebula) give $M_\ast > 4.1 \times 10^{14}$\,GeV, i.e.\ $\ell_\ast < 4.82 \times 10^{-31}$\,m, and there $|\alpha_t| = x/(1+x)$, $x = (H_{\mathrm{inf}}/M_\ast)^2$, is at most of order $10^{-2}$.
\end{conceptbox}

\begin{conceptbox}{{8. Functional Realizations, Cut Norms and Hypergraphons (Ch. 1)}}
\textbf{Cut norm (proved, Ch.~1):} $\|W\|_\square \le \|T_W\|_{L^\infty \to L^1} \le 4\|W\|_\square$. For $k$-hypergraphons, $\|W\|_{\square,k} \le \|T_W\|_{L^\infty \times \dots \times L^\infty \to \mathbb{R}} \le 2^k\|W\|_{\square,k}$; the proof passes to indicator functions one factor at a time, splitting each test function into positive and negative parts. The quotient space $(\widetilde{\mathcal{W}}_k, \delta_{\square,k})$ is compact.\\
\textbf{Spectral blindness (proved, Ch.~1):} the Dirichlet energy of a matrix realization is not a spectral invariant: permutation conjugation keeps the spectrum and the Frobenius norm, yet the energy ratio of two such conjugates can be of order $n^2$.\\
\textbf{Tensors (proved, Ch.~1):} the best rank-one approximation exists on the product of spheres, and its critical points satisfy Lim's singular-vector equations with a common value $\sigma$; Gaussian tensors require the normalization $d^{-1/2}$ (critical scaling).
\end{conceptbox}

\begin{conceptbox}{{9. Flows on Matrix and Tensor Manifolds (Ch. 2)}}
\textbf{Toda flow and QR (proved, Ch.~2):} $A' = [A, \Pi_{\mathfrak{so}}(A)]$ has a unique global symmetric solution, is isospectral, and $A(t) = Q(t)^T A_0 Q(t)$, where $\exp(tA_0) = Q(t)R(t)$ is the QR factorization with $R(t)$ upper triangular with positive diagonal (unique and smooth in $t$). At integer times the flow reproduces the unshifted QR iterates of $e^{A_0}$, not of $A_0$.\\
\textbf{Fixed-rank manifold (Ch.~2, with the tangent-space description cited from Vandereycken 2013):} the orthogonal projection onto $T_A\mathcal{M}_r$ is $P(Z) = Z - (I - UU^T)Z(I - VV^T)$; projected gradient flows decrease the loss.\\
\textbf{Graphon heat equation (proved, Ch.~2):} closed-form solution $W(t) = e^{-2t}W_0 + (e^{-t} - e^{-2t})(d_1 + d_2) + (1 - e^{-t})^2 m$; the cut norm contracts; jumps between cells with equal degrees persist with amplitude $e^{-2t}$.\\
\textbf{Path integral on a loop (proved, Ch.~2):} $Z_k \to \operatorname{Tr}\mathcal{P}\exp\oint\mathcal{A}$ with error at most $\frac{r}{k}(M + \frac{a_1 + a_0^2}{2})\exp(\|\mathcal{A}\|_{L^1} + \frac{M + a_1/2}{k})$, where $M$ bounds the remainders $\mathcal{R}_j$. The factor $e^{M/k}$ cannot be dropped: for $r = 1$, $\mathcal{A} \equiv 0$ and $\mathcal{R}_j \equiv M > 0$ the error is $(1 + M/k^2)^k - 1 > M/k$ for $k \ge 2$. The trace of the holonomy is invariant under periodic gauge transformations $\mathcal{A} \mapsto \Omega^{-1}\mathcal{A}\Omega - \Omega^{-1}\Omega'$.
\end{conceptbox}

\begin{conceptbox}{{10. Continuous Pascal Simplex (Ch. 3, 6)}}
\textbf{Identities (proved, Ch.~3):} Stifel recurrence $\binom{x-1}{y} + \binom{x-1}{y-1} = \binom{x}{y}$; the continuous Star of David identity; vanishing on $y \in \mathbb{Z}_{<0}$ and $x - y \in \mathbb{Z}_{<0}$ and positivity for $x > 0$, $0 \le y \le x$.\\
\textbf{Digamma system (proved, Ch.~3):} on the open set where none of $x+1$, $y+1$, $x-y+1$ lies in $\mathbb{Z}_{\le 0}$, $\partial_x\binom{x}{y} = \binom{x}{y}[\psi(x+1) - \psi(x-y+1)]$ and $\partial_y\binom{x}{y} = \binom{x}{y}[\psi(x-y+1) - \psi(y+1)]$; each product extends holomorphically across the zero lines.\\
\textbf{Asymptotics (proved, Ch.~3):} $I_m(x) \sim m^x$; the continuous Fibonacci diagonal $\int_0^{x/2}\binom{x-y}{y}\,dy$ grows like $\phi^{x+1}/\sqrt5$, with saddle point $\xi^* = (5 - \sqrt5)/10$; the continuous Dixon integral is evaluated for integer $x$; the row entropy has a closed form through the Barnes $G$-function and the asymptotic expansion of Ch.~6.\\
\textbf{Sierpi\'nski (Ch.~6):} the harmonic decimation factor is $r_m = m+3$ (proved); Pascal's triangle modulo $2$ has box-counting dimension $\ln 3/\ln 2$ (classical).
\end{conceptbox}

\begin{conceptbox}{{11. Nonlinear Simplicial Waves and Anomalous Transport (Ch. 4)}}
\textbf{Conservation (proved, Ch.~4):} the nonlocal NLSE with the simplicial Laplacian conserves mass and energy.\\
\textbf{Modulational instability (proved, Ch.~4):} $\hbar^2\Omega^2 = \frac{\hbar^2\sigma}{2M}\big(\frac{\hbar^2\sigma}{2M} - 2\kappa p\rho_0^p\big)$; in the focusing regime a plane wave is unstable exactly for the wavevectors with $0 < \sigma_{\Delta_m}^\alpha(\mathbf{k}) < 4M\kappa p\rho_0^p/\hbar^2$, and $\Omega = 0$ where $\sigma = 0$, in particular at $\mathbf{k} = \mathbf{0}$. The growth rate is at most $\kappa p\rho_0^p/\hbar$, attained on the resonant set $\{\sigma = 2M\kappa p\rho_0^p/\hbar^2\}$ when that set is non-empty. Because the symbol is bounded, every $\mathbf{k} \ne \mathbf{0}$ is unstable when $4M\kappa p\rho_0^p/\hbar^2 > \sup\sigma$.\\
\textbf{Fractional diffusion (proved, Ch.~4):} $\widehat u(\mathbf{k}, t) = E_\beta(-\mathcal{K}_{\mathrm{diff}}\sigma(\mathbf{k})t^\beta)\widehat u_0$; the mean squared displacement is $\frac{\mathcal{K}_{\mathrm{diff}}}{\alpha^2\Gamma(\beta+1)}\mathbf{M}_2(\alpha)\,t^\beta$ exactly.
\end{conceptbox}

\begin{conceptbox}{{12. Curvature-Rate Routing, Wall Contact and Squeezing (Ch. 7)}}
\textbf{Wall problem (proved, Ch.~7, structure lemma):} for the linear wall problem $P_W^T(\xi)$ the minimiser is unique; once the state reaches the origin it rests there; boundary arcs occur only on the rest interval $[t_*, T]$; every contact point before $t_*$ is isolated, with $\theta = 0 < \kappa$, and contacts can accumulate only at $t_*$. \textbf{Conjecture (Ch.~7):} the contact set is finite, and a related statement for the length cost $Q_B^T$.\\
\textbf{Chattering (classical, cited in Ch.~7):} shortest $C^2$ paths with $|\kappa'| \le B$ that contain a straight segment carry infinitely many clothoid arcs accumulating at its ends (Fuller's phenomenon; Boissonnat--Cerezo--Leblond, Sussmann; optimality for generic data as stated by Kostov and Degtiariova-Kostova). \textbf{Numerical observation:} in Fuller's problem the excess cost of the best control with $n$ switches decays like $c\,k^{5n}$, $k \approx 0.2421$; a matching upper bound is sketched, the lower bound is open.\\
\textbf{Ball bound (proved, Ch.~7):} a compact $C^{1,1}$ immersed submanifold without boundary with $\|\II\|_{L^\infty} \le \kappa$ inside a ball of radius $R$ has $\kappa R \ge 1$, with equality for round spheres. Hence the $C^1$ isometric embeddings of Nash--Kuiper type, which fit in arbitrarily small balls, are not competitors, and an $L^\infty$ bound on $\II$ excludes corrugated microstructure. The closed curve with tangent angle $\theta + \beta\sin(N\theta)$ is closed for every $\beta > 0$ and $N \ge 2$ (Jacobi--Anger expansion), and has curvature $1 + \beta N$.
\end{conceptbox}

\begin{conceptbox}{{13. Corridors, Mouths and the Teardrop U-Turn (Ch. 9)}}
\textbf{Setting:} a corridor ends in a mouth $W_a$ of width $a$ opening into a balloon; admissible curves are $C^{1,1}$ with $|\kappa| \le \kappa^* = 1/\rho$.\\
\textbf{Proved (Ch.~9):} a reversal with tangent angle in $[0,\pi]$ needs $a \ge 2\rho$; any reversal spans a height $2\rho$, so a mouth U-turn inside $D_R$ needs $R \ge \rho$. For $a < 2\rho$ the three-arc teardrop $\gamma_a$ (right turn $\beta$, left turn $\pi + 2\beta$, right turn $\beta$) is an embedded mouth U-turn with far distance $r_*(a) = \rho(1 + 2\sin\beta(a))$, and it is the unique minimizer of the far distance among three-arc curves of the two turning patterns (up to reflection). An embedded admissible curve in $D_R \cap \{x \ge 0\}$ with end points $q_0 \ne q_1$ in $W_a$ that meets the disc with diameter $[q_0, q_1]$ only at its end points satisfies $R \ge \rho + \sqrt{\rho^2 - a^2/4}$. The formula for $r_*(a)$ is the classical depth of the Omega-turn of headland path planning.\\
\textbf{Conjecture (Ch.~9):} every embedded mouth U-turn has $\max|\gamma| \ge r_*(a)$.
\end{conceptbox}

\begin{conceptbox}{{14. Information Geometry (Ch. 10, 11)}}
\textbf{Degeneracy (proved, Ch.~10):} $\kappa^*_{\mathrm{info}} = 0$ whenever an admissible geodesic reaches the target set.\\
\textbf{Counterexample (proved, Ch.~10):} for Gaussian linear regression with $X \in \mathbb{R}^{N\times D}$ of full row rank $N$, $D > N$, the minimum-norm interpolant has $\kappa^*_{\mathrm{info}} = 0$, positive Hessian trace, and generalization gap at least $1/2$ (uniform $\pm1$ labels, squared loss clipped to $[0,1]$) for every input distribution under which $X$ has rank $N$ almost surely. Without the rank hypothesis the conclusion can fail ($x \equiv 0$).\\
\textbf{Quantum Fisher metrics (Ch.~11):} the Hessian of relative entropy is $g^{\mathrm{BKM}}$; $g^{\mathrm{BKM}}(X,X) \ge g^{\mathrm{SLD}}(X,X)$, with equality iff $[\rho_0, X] = 0$, in which case both equal $\operatorname{Tr}(\rho_0^{-1}X^2)$, the classical Fisher--Rao value $\sum_i X_{ii}^2/p_i$ in an eigenbasis of $\rho_0$ that also diagonalizes $X$. The ratio $g^{\mathrm{BKM}}/g^{\mathrm{SLD}}$ is unbounded.\\
\textbf{Tensor networks (proved, Ch.~11):} min-cut bound $S_A \le \log\chi\cdot\min_C|C|$; first law $\delta S_A = \delta\langle H_A\rangle$. The emergence of the linearized Einstein equations from entanglement is the cited theorem of Faulkner--Guica--Hartman--Myers--Van Raamsdonk.
\end{conceptbox}
