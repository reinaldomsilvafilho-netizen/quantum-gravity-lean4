\subsection{Critical Scaling and Measure Concentration in Infinite Dimensions ($d \to \infty$)}
\label{subsec:measure_concentration}

A central dilemma in high-dimensional tensor analysis is establishing the exact normalization that prevents multilinear potentials from either exploding or vanishing as the mode dimension $d \to \infty$.

\begin{theorem}[Critical Multilinear Scaling and Sub-Gaussian Concentration]
\label{thm:critical_scaling}
Let $\mathcal{T} \in (\R^d)^{\otimes k}$ be a random tensor whose entries are centered, independent, and identically distributed with unit variance and a sub-Gaussian tail. Define the normalized functional realization:
\begin{equation}
    \label{eq:renorm_tensor_form}
    \widehat{f}_{\mathcal{T}}(u^{(1)}, \dots, u^{(k)}) \coloneqq \frac{1}{\sqrt{d}} \sum_{j_1, \dots, j_k = 1}^d \mathcal{T}_{j_1 \dots j_k} u_{j_1}^{(1)} \dots u_{j_k}^{(k)}, \quad u^{(\alpha)} \in \Sph^{d-1}.
\end{equation}
Then:
\begin{enumerate}[label=(\roman*)]
    \item The normalization exponent $\alpha_k = \tfrac{1}{2}$, \emph{independent of $k$}, is the unique critical scaling ensuring that the operator norm remains bounded and non-trivial:
    \begin{equation}
        0 < \liminf_{d\to\infty}\mathbb{E}\left[ \sup_{u^{(\alpha)} \in \Sph^{d-1}} |\widehat{f}_{\mathcal{T}}(u^{(1)}, \dots, u^{(k)})| \right] \le \limsup_{d\to\infty}\mathbb{E}\left[ \sup_{u^{(\alpha)} \in \Sph^{d-1}} |\widehat{f}_{\mathcal{T}}(u^{(1)}, \dots, u^{(k)})| \right] < \infty,
    \end{equation}
    whereas for every $\beta \ne 1/2$ the rescaled functional $d^{1/2-\beta}\widehat f_{\mathcal T}$ satisfies $\mathbb E[\sup|\cdot|] \to 0$ if $\beta<1/2$ and $\to\infty$ if $\beta>1/2$. (This exponent coincides with the classical Ginibre operator-norm exponent at $k=2$; for $k\ge 3$ the constant, not the exponent, grows with $k$ — cf.\ \cite{tomioka2014spectral,nguyen2015tensor}.)
    \item For any independent tuple of random vectors $(u^{(1)}, \dots, u^{(k)})$ drawn uniformly from the Haar measure on $\Sph^{d-1}$, the functional satisfies the sharp sub-Gaussian concentration inequality:
    \begin{equation}
        \label{eq:concentration_inequality}
        \mathbb{P}\left( |\widehat{f}_{\mathcal{T}}(u^{(1)}, \dots, u^{(k)})| > \epsilon \right) \le 2 \exp\left( - c_k \, d \, \epsilon^2 \right),
    \end{equation}
    where $c_k > 0$ is a universal dimensional constant depending only on $k$.
\end{enumerate}
\end{theorem}
\begin{proof}
Write $X(u) \coloneqq \sum_{j_1,\dots,j_k} \mathcal{T}_{j_1\dots j_k} u_{j_1}^{(1)}\cdots u_{j_k}^{(k)}$, so $\widehat f_{\mathcal T}=d^{-1/2}X$, and let $T \coloneqq \prod_{\alpha=1}^k \Sph^{d-1}$, a manifold of dimension $k(d-1)$. For each fixed $u\in T$, $X(u)$ is a linear functional of the (sub-)Gaussian vector $\mathcal T$, so $\{X(u)\}_{u\in T}$ is a genuine centered (sub-)Gaussian process; as computed above, $\operatorname{Var}(X(u)) = \prod_{\alpha=1}^k\|u^{(\alpha)}\|_2^2 = 1$ for every $u\in T$, and by the same computation the canonical increment metric satisfies, for $u,u'$ differing only in coordinate $\alpha$,
\begin{equation}
    \mathbb E\bigl[(X(u)-X(u'))^2\bigr] = \Bigl\|u^{(\alpha)}-u'^{(\alpha)}\Bigr\|_2^2 \prod_{\beta\ne\alpha}\|u^{(\beta)}\|_2^2 = \bigl\|u^{(\alpha)}-u'^{(\alpha)}\bigr\|_2^2,
\end{equation}
so the process is $1$-Lipschitz with respect to the product metric $d_T(u,u')=\bigl(\sum_\alpha\|u^{(\alpha)}-u'^{(\alpha)}\|_2^2\bigr)^{1/2}$ on $T\subset(\R^d)^{k}$.

\emph{Upper bound.} $(\Sph^{d-1},\|\cdot\|_2)$ admits an $\eta$-net of cardinality $N(\eta)\le(3/\eta)^{d-1}$ for $\eta\in(0,1)$, so $T$ admits an $\eta$-net (in $d_T$) of cardinality $N_T(\eta)\le(3/\eta)^{k(d-1)}$. Dudley's entropy integral for the sub-Gaussian process $X$ gives
\begin{equation}
    \mathbb E\Bigl[\sup_{u\in T}|X(u)|\Bigr] \le C\int_0^{\sqrt{2k}}\sqrt{\log N_T(\eta)}\,\dif\eta \le C' \sqrt{k(d-1)} = O(\sqrt{kd}).
\end{equation}
Hence $\mathbb E[\sup_T|\widehat f_{\mathcal T}|] = d^{-1/2}O(\sqrt{kd}) = O(\sqrt k)$, uniformly in $d$.

\emph{Matching lower bound.} Fix $u^{(2)},\dots,u^{(k)}$ and vary only $u^{(1)}$: $u^{(1)}\mapsto X(u)$ is then \emph{linear} with $\mathbb E\|\nabla_{u^{(1)}} X\|_2^2 = \mathbb E\Bigl\|\sum_{j_2,\dots,j_k}\mathcal T_{\cdot,j_2\dots j_k}u^{(2)}_{j_2}\cdots u^{(k)}_{j_k}\Bigr\|_2^2 = d$ (a sum of $d$ i.i.d.\ unit-variance coordinates by the variance identity above), so with probability $\to 1$ this gradient has Euclidean norm $\Theta(\sqrt d)$. Restricting the supremum over $u^{(1)}\in\Sph^{d-1}$ alone to the linear functional generated by this gradient gives $\sup_{u^{(1)}}|X(u)|\ge (1-o(1))\sqrt d$ with probability $\to1$, hence $\mathbb E[\sup_T|X|] \ge c\sqrt d$ for an absolute constant $c>0$, i.e.\ $\mathbb E[\sup_T|\widehat f_{\mathcal T}|]\ge c>0$ for all $d$.

Combining the two bounds, $0<\liminf_d \mathbb E[\sup_T|\widehat f_{\mathcal T}|]\le\limsup_d \mathbb E[\sup_T|\widehat f_{\mathcal T}|]<\infty$, proving the two-sided estimate in (i). For $\beta\ne1/2$, $d^{1/2-\beta}\widehat f_{\mathcal T} = d^{-\beta}X$ inherits this same two-sided $\Theta(1)$ bound on $X$'s sup, multiplied by $d^{1/2-\beta}\to0$ (if $\beta>1/2$) or $\to\infty$ (if $\beta<1/2$), so $\alpha=1/2$ is the unique exponent for which the limit is neither $0$ nor $\infty$.

To establish (ii), recall that the round sphere $\Sph^{d-1}$ has Ricci curvature $\mathrm{Ric}(\Sph^{d-1}) = (d-2) g_{\Sph}$. By the L\'evy--Gromov isoperimetric theorem and the Bakry--\'Emery criterion, any Lipschitz function $F \colon \Sph^{d-1} \to \R$ with Lipschitz constant $\|F\|_{\Lip} \le L$ satisfies:
\begin{equation}
    \mathbb{P}\left( |F - \mathbb{E}[F]| > \epsilon \right) \le 2 \exp\left( - \frac{(d-1)\epsilon^2}{2 L^2} \right).
\end{equation}
Conditioning on $(u^{(2)}, \dots, u^{(k)})$, the function $u^{(1)} \mapsto \widehat{f}_{\mathcal{T}}(u^{(1)}, \dots, u^{(k)})$ is linear with Lipschitz constant:
\begin{equation}
    L_1 = \frac{1}{\sqrt d} \left\| \sum_{j_2, \dots, j_k} \mathcal{T}_{:, j_2 \dots j_k} u_{j_2}^{(2)} \dots u_{j_k}^{(k)} \right\|_2,
\end{equation}
which by the gradient-norm identity used in the lower bound above satisfies $\mathbb E[L_1^2] = d^{-1}\cdot d = 1$, so $L_1 = \Theta(1)$ with overwhelming probability — the correct normalization $\alpha=1/2$ is exactly the one for which $L_1$ neither degenerates to $0$ nor diverges, making the bound below non-trivial. Taking the product expectation and applying the Herbst argument inductively across all $k$ spheres yields \eqref{eq:concentration_inequality} with $c_k = \Theta(1/k)$.
\end{proof}
